% !tp-example Basic usage
\documentclass{article}
\begin{document}
\wcTpHtmlViewer{
  \wcScript[type=tp/html]|
<h2>Hello, HTML!</h2>
<p>This preview renders <strong>HTML content</strong>.</p>
  |
}
\end{document}

        </div>

}
