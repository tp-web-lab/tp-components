% !tp-example Basic usage
\documentclass{article}
\begin{document}
\wcTpHtmlViewer{
  \wcScript[type=tp/html]|
<h2>Hello, HTML!</h2>
<p>This preview renders <strong>HTML content</strong>.</p>
  |
}
\end{document}

% !tp-example Javascript API
\documentclass{article}
\begin{document}
\wcScript[type=module]|
import "/components/html-viewer/html-viewer.js";

const example = document.createDocumentFragment();

const element1 = document.createElement("tp-html-viewer");
element1.append(document.createTextNode("\n  "));

const element2 = document.createElement("script");
element2.setAttribute("type", "tp/html");
element2.textContent = "\n\u003ch2>Hello, HTML!\u003c/h2>\n\u003cp>This preview renders \u003cstrong>HTML content\u003c/strong>.\u003c/p>\n  ";
element1.append(element2);
element1.append(document.createTextNode("\n"));
example.append(element1);

document.body.append(example);
|
\end{document}

% !tp-example Compare presentation modes
\documentclass{article}
\begin{document}
\wcSection{
  \subsection*{mode="default" — Standard controls}
  Open Code to edit the source, then click Run or press \wcKbd*{Shift}+\wcKbd*{Enter} in the editor.

  \wcTpHtmlViewer[mode=default]{
    \wcScript[type=tp/html]|
<h2>Viewer modes</h2>
<p>Change this message, then run the source.</p>
  |
  }
}

\wcSection{
  \subsection*{mode="lite" — Compact controls}
  Open Code to edit the source, then click Run or press \wcKbd*{Shift}+\wcKbd*{Enter} in the editor.

  \wcTpHtmlViewer[mode=lite]{
    \wcScript[type=tp/html]|
<h2>Viewer modes</h2>
<p>Change this message, then run the source.</p>
  |
  }
}

\wcSection{
  \subsection*{mode="simple" — Editor and rendered output only}
  Edit the code below, keep the cursor in the editor and press \wcKbd*{Shift}+\wcKbd*{Enter} (Return) to run it. This viewer has no Run button.

  \wcTpHtmlViewer[mode=simple]{
    \wcScript[type=tp/html]|
<h2>Viewer modes</h2>
<p>Change this message, then run the source.</p>
  |
  }
}
\end{document}
