\documentclass{article}
\begin{document}
\wcScript[type=module]|

const example = document.createDocumentFragment();

element21.setAttribute("style", "vertical-align:middle");

const element22 = document.createElement("td");
element22.setAttribute("data-type", "math");
element22.append(document.createTextNode("E = mc^2"));

element9.append(element16);
element9.append(document.createTextNode("\n    "));
element2.append(element9);
element2.append(document.createTextNode("\n  "));
element1.append(element2);
element1.append(document.createTextNode("\n"));
example.append(element1);

document.body.append(example);
|

% !tp-example Rich content
\documentclass{article}
\begin{document}
\wcTpTableEditor[caption=Rich content export comparison]{
  \wcTable{
    \wcThead{
      \wcTr{
        \wcTh{Example}
        \wcTh{Content}
      }
    }
    \wcTbody{
      \wcTr{
        \wcTd{Bold text}
        \wcTd[data-type=rich]{
          \textbf{Compare bold text in each export.}

        }
      }
      \wcTr{
        \wcTd{Image}
        \wcTd[data-type=rich]{
          \wcImg*[src=/medias/image/canoe.svg,alt=Canoe,width=160]{}
        }
      }
      \wcTr{
        \wcTd{Video}
        \wcTd[data-type=rich]{
          \wcVideo[src=/medias/video/chute-eau.mp4,controls,preload=metadata,width=240,aria-label=Waterfall video]{}
        }
      }
      \wcTr{
        \wcTd{Audio}
        \wcTd[data-type=rich]{
          \wcAudio[src=/medias/audio/glass-break.mp3,controls,preload=metadata,aria-label=Glass breaking sound]{}
        }
      }
    }
  }
}
\end{document}
