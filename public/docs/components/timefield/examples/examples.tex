% !tp-example Basic usage
\documentclass{article}
\begin{document}
\wcTpTimefield[label=Time,value=14:30,clearable]{}
\end{document}

% !tp-example Javascript API
\documentclass{article}
\begin{document}
\wcScript[type=module]|
import "/components/timefield/timefield.js";

const example = document.createDocumentFragment();

const element1 = document.createElement("tp-timefield");
element1.setAttribute("label", "Time");
element1.setAttribute("value", "14:30");
element1.setAttribute("clearable", "");
example.append(element1);

document.body.append(example);
|
\end{document}

% !tp-example Inline named field
\documentclass{article}
\begin{document}

\end{document}
