% !tp-example Custom style
\documentclass{article}
\begin{document}
\rawhtml|<style>
      .divider-custom-style tp-divider {
        --tp-divider-color: var(--tp-brand-fill-mid);
        --tp-divider-thickness: 4px;
        --tp-divider-margin-block: 0.75rem;
      }
    </style>|

\wcTpBox{
  \wcDiv[class=divider-custom-style]{
    Before the custom divider.

    \wcTpDivider{}
  }
  After the custom divider.

}
\end{document}
