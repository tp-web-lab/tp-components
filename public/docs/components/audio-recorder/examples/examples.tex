% !tp-example Spoken answer
\documentclass{article}
\begin{document}
\wcTpFillBlankQuestion[open]{
  \begin{description}
    \item[{Title}] A spoken answer
    \item[{Answers}] \begin{enumerate}
      \item blue
    \end{enumerate}
    \item[{Prompt}] Say the word “blue”. Stop recording, review the transcript and submit. Recognition may use an online service; you can also type the answer.
    \item[{Form}] \rawhtml|<tp-audio-recorder transcript="#spoken-answer" lang="en-US"></tp-audio-recorder>|     \rawhtml|<tp-textfield id="spoken-answer" name="color" clearable="" regexp="blue[.!?]?" label="Your answer"></tp-textfield>|
    \item[{Solution}] blue
  \end{description}
}
\end{document}
