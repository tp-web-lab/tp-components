% !tp-example Selection event
\documentclass{article}
\begin{document}
\wcTpFormulaPicker[id=formula-event-picker]{}

\wcTpTextfield[id=formula-event-result,label=Selected formula,placeholder=Select a formula,clearable]{}

\wcScript[type=module]|const picker = document.querySelector("#formula-event-picker");
const result = document.querySelector("#formula-event-result");
await customElements.whenDefined("tp-textfield");
picker.addEventListener("tp-formula-picker-select", (event) => {
  result.value = event.detail.formula;
});|
\end{document}
