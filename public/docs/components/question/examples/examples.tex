% !tp-example Basic usage
\documentclass{article}
\begin{document}
\wcTpQuestion{
  \begin{description}
    \item[{Title}] Reflection
    \item[{Prompt}] Describe one benefit of web components.
    \item[{Form}] \wcTpTextfield[multiline,rows=3,placeholder=Type your answer...]{}
    \item[{Feedback}] Think about encapsulation and reuse.
    \item[{Solution}] They package reusable behavior behind a custom element.
  \end{description}
}
\end{document}

% !tp-example Timed question
\documentclass{article}
\begin{document}
\wcTpQuestion[timing=00:00:10,open]{
  \begin{description}
    \item[{Title}] A timed reflection
    \item[{Prompt}] Name one reusable web component.
    \item[{Form}] \wcTpTextfield[clearable,placeholder=Your answer]{}
    \item[{Solution}] A button, a timer or an audio recorder.
  \end{description}
}
\end{document}

% !tp-example Optional score
\documentclass{article}
\begin{document}
\wcTpFillBlankQuestion[score,open]{
  \begin{description}
    \item[{Title}] An optional numerical score
    \item[{Answers}] \begin{enumerate}
      \item blue
    \end{enumerate}
    \item[{Prompt}] Type blue, then submit.
    \item[{Form}] \wcTpTextfield[clearable,label=Color]{}
    \item[{Feedback}] Use the word requested in the prompt.
    \item[{Solution}] blue
  \end{description}
}
\end{document}
