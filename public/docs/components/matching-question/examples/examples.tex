% !tp-example Basic usage
\documentclass{article}
\begin{document}
\wcTpMatchingQuestion{
  \begin{description}
    \item[{Title}] English and French
    \item[{Prompt}] Match each English expression with its French translation.
    \item[{Form}] \begin{itemize}
      \item Hello
      \item Thank you
      \item Goodbye
    \end{itemize}     \begin{enumerate}
      \item Bonjour
      \item Merci
      \item Au revoir
    \end{enumerate}
    \item[{Feedback}] Distinguish greetings, thanks and farewells.
    \item[{Solution}] Hello — Bonjour; Thank you — Merci; Goodbye — Au revoir.
  \end{description}
}
\end{document}

      \item Breaking glass
      \item A waterfall

    \item[{Solution}] The flag represents Lithuania; the sound is breaking glass; the video shows a waterfall.
  \end{description}
}

% !tp-example Irregular verbs
\documentclass{article}
\begin{document}
\wcTpMatchingQuestion{
  \begin{description}
    \item[{Title}] Irregular verbs
    \item[{Prompt}] Associate each verb with its past simple, past participle and French translation.
    \item[{Form}] \wcTpMatching{
      \begin{description}
        \item[{Base form}] \begin{itemize}
          \item be
          \item have
          \item do
        \end{itemize}
        \item[{Past simple}] \begin{enumerate}
          \item was
          \item had
          \item did
        \end{enumerate}
        \item[{Past participle}] \begin{itemize}
          \item been
          \item had
          \item done
        \end{itemize}
        \item[{French}] \begin{enumerate}
          \item être
          \item avoir
          \item faire
        \end{enumerate}
      \end{description}
    }
    \item[{Feedback}] Check all four members of each group before submitting again.
    \item[{Solution}] \begin{itemize}
      \item be — was — been — être
      \item have — had — had — avoir
      \item do — did — done — faire
    \end{itemize}
  \end{description}
}
\end{document}
