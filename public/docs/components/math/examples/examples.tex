% !tp-example Script and display style
\documentclass{article}
\begin{document}

  \wcScript[type=tp/math]|\frac{1}{2\sqrt{x}}|
}

\wcTpMath[displaystyle]{
  \wcScript[type=tp/math]|\frac{1}{2\sqrt{x}}|
}
\end{document}
