% !tp-example Basic usage
\documentclass{article}
\begin{document}
\wcTpInclude[src=/docs/components/include/examples/welcome.html,loading=Loading the shared notice…]{}
\end{document}

% !tp-example Raw HTML
\documentclass{article}
\begin{document}
\wcTpInclude[src=/docs/components/include/examples/welcome.html,mode=raw]{}
\end{document}

% !tp-example Javascript API
\documentclass{article}
\begin{document}
\wcScript[type=module]|
import "/components/include/include.js";

const example = document.createDocumentFragment();

const element1 = document.createElement("tp-include");
element1.setAttribute("src", "/docs/components/include/examples/welcome.html");
element1.setAttribute("loading", "Loading the shared notice…");
example.append(element1);

document.body.append(example);
|
\end{document}
