\documentclass{article}
\begin{document}
\wcTpObjectTree{
  \wcScript[type=tp/javascript]|
    ({
      course: {
        title: 'Web components',
        lessons: 12,
        published: true
      },
      topics: ['HTML', 'CSS', 'JavaScript']
    })
  |
}
\end{document}

        ({
          component: 'tp-object-tree',
          stable: true,
          versions: [1, 2, 3]
        })
      |
}
\end{document}

% !tp-example JavaScript API
\documentclass{article}
\begin{document}
\wcTpObjectTree{}

\wcScript|
      {
      const script = document.currentScript;
      const output = script?.closest('[data-role="output"]');
      const tree = output?.querySelector('tp-object-tree')
        ?? script?.previousElementSibling;

      customElements.whenDefined('tp-object-tree').then(() => {
        tree?.setValue({
          component: 'tp-object-tree',
          stable: true,
          versions: [1, 2, 3]
        });
      });
      }
    |
\end{document}
