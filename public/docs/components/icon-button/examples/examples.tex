  Activate the heart button to add a like.

  \wcTpIconButton[name=heart,label=Like this example,data-demo-trigger]{}
  \wcP[data-demo-status,role=status]{Likes: 0}
  \wcScript[src=/docs/components/_shared/introduction-actions.js]||

% !tp-example Javascript API
\documentclass{article}
\begin{document}
\wcScript[type=module]|
import "/components/box/box.js";
import "/components/icon/icon.js";
import "/components/icon-button/icon-button.js";

const example = document.createDocumentFragment();

const element1 = document.createElement("tp-box");
element1.setAttribute("data-intro-action", "counter");
element1.setAttribute("data-allow-script", "");
element1.append(document.createTextNode("\n  "));

const element2 = document.createElement("p");
element2.append(document.createTextNode("Activate the heart button to add a like."));
element1.append(element2);
element1.append(document.createTextNode("\n  "));

const element3 = document.createElement("tp-icon-button");
element3.setAttribute("name", "heart");
element3.setAttribute("label", "Like this example");
element3.setAttribute("data-demo-trigger", "");
element1.append(element3);
element1.append(document.createTextNode("\n  "));

const element4 = document.createElement("p");
element4.setAttribute("data-demo-status", "");
element4.setAttribute("role", "status");
element4.append(document.createTextNode("Likes: 0"));
element1.append(element4);
element1.append(document.createTextNode("\n  "));

const element5 = document.createElement("script");
element5.setAttribute("src", "/docs/components/_shared/introduction-actions.js");
element1.append(element5);
element1.append(document.createTextNode("\n"));
example.append(element1);

document.body.append(example);
|
\end{document}
