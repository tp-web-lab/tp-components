% !tp-example Basic usage
\documentclass{article}
\begin{document}
\wcTpRestructuredtext{
  \wcScript[type=tp/restructuredtext]|
    Hello, **Docutils**!
  |
}
\end{document}

% !tp-example Javascript API
\documentclass{article}
\begin{document}
\wcScript[type=module]|
import "/components/restructuredtext/restructuredtext.js";

const example = document.createDocumentFragment();

const element1 = document.createElement("tp-restructuredtext");
element1.append(document.createTextNode("\n  "));

const element2 = document.createElement("script");
element2.setAttribute("type", "tp/restructuredtext");
element2.textContent = "\n    Hello, **Docutils**!\n  ";
element1.append(element2);
element1.append(document.createTextNode("\n"));
example.append(element1);

document.body.append(example);
|
\end{document}
