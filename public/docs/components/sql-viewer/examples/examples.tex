% !tp-example script without filename
\documentclass{article}
\begin{document}
\wcTpSqlViewer{
  \wcScript[type=tp/sql]|
        CREATE TABLE users (name TEXT NOT NULL);
        INSERT INTO users (name) VALUES ('Ada'), ('Grace');
      |
}
\end{document}

% !tp-example scripts with filename
\documentclass{article}
\begin{document}
\wcTpSqlViewer{
  \wcScript[type=tp/sql,filename=tables.sql]|
        CREATE TABLE users (name TEXT NOT NULL);
        INSERT INTO users (name) VALUES ('Ada'), ('Grace');
      |
  \wcScript[type=tp/sql,filename=query.sql]|
        SELECT name
        FROM users
        ORDER BY name;
      |
}
\end{document}
