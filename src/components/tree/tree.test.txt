/**
 * Attend la fin des micro-tâches en cours.
 *
 * @summary Attend la stabilisation asynchrone du DOM.
 * @returns Promesse résolue au prochain tour de micro-tâche.
 */
async function flush(): Promise<void> {
  await Promise.resolve();
}

/**
 * Retourne une valeur définie ou lève une erreur explicite.
 *
 * @summary Garantit qu'une valeur n'est ni `null` ni `undefined`.
 * @param value Valeur à vérifier.
 * @param message Message d'erreur si la valeur est absente.
 * @returns Valeur non nulle.
 * @throws {Error} Si la valeur est absente.
 */
function expectDefined<T>(value: T | null | undefined, message: string): T {
  if (value == null) {
    throw new Error(message);
  }

  return value;
}

/**
 * Retourne le premier élément `<tp-tree>` du document.
 *
 * @summary Récupère l’élément arbre de test.
 * @returns Élément `<tp-tree>`.
 * @throws {Error} Si aucun arbre n’est trouvé.
 */
function getTree(): TpTree {
  const element = document.querySelector('tp-tree');

  if (!(element instanceof HTMLElement)) {
    throw new Error('tp-tree not found');
  }

  return element as TpTree;
}

function getDragdrop(): HTMLElement {
  const element = document.querySelector('tp-tree tp-dragdrop');

  if (!(element instanceof HTMLElement)) {
    throw new Error('tp-dragdrop not found inside tp-tree');
  }

  return element;
}

/**
 * Retourne tous les nœuds `<li>` du premier arbre.
 *
 * @summary Récupère tous les nœuds de l’arbre.
 * @returns Liste des nœuds.
 */
function getItems(): HTMLLIElement[] {
  const tree = getTree();

  return Array.from(tree.querySelectorAll('li')).filter(
    (element): element is HTMLLIElement => element instanceof HTMLLIElement,
  );
}

/**
 * Retourne les nœuds racine directs.
 *
 * @summary Récupère les nœuds directs de la racine.
 * @returns Liste des nœuds racine.
 */
function getRootItems(): HTMLLIElement[] {
  const tree = getTree();
  const rootList = tree.querySelector(':scope > ul');

  if (!(rootList instanceof HTMLUListElement)) {
    throw new Error('root list not found');
  }

  return Array.from(rootList.children).filter(
    (element): element is HTMLLIElement => element instanceof HTMLLIElement,
  );
}

/**
 * Retourne le premier bouton de bascule d’un nœud.
 *
 * @summary Récupère le bouton toggle d’un nœud.
 * @param item Nœud concerné.
 * @returns Bouton toggle ou `null`.
 */
function getToggle(item: HTMLLIElement): HTMLButtonElement | null {
  const button = item.querySelector(':scope > [data-tp-tree-toggle]');
  return button instanceof HTMLButtonElement ? button : null;
}

/**
 * Retourne le premier libellé d’un nœud.
 *
 * @summary Récupère le libellé d’un nœud.
 * @param item Nœud concerné.
 * @returns Élément de libellé ou `null`.
 */
function getLabel(item: HTMLLIElement): HTMLElement | null {
  const label = item.querySelector(':scope > [data-tp-tree-label]');
  return label instanceof HTMLElement ? label : null;
}

/**
 * Retourne le texte du libellé d’un nœud.
 *
 * @summary Retourne le texte lisible d’un nœud.
 * @param item Nœud concerné.
 * @returns Texte du nœud.
 */
function getItemText(item: HTMLLIElement): string {
  return getLabel(item)?.textContent?.trim() ?? '';
}

/**
 * Retourne la première sous-liste directe d’un nœud.
 *
 * @summary Récupère la sous-liste d’un nœud.
 * @param item Nœud concerné.
 * @returns Sous-liste ou `null`.
 */
function getChildList(item: HTMLLIElement): HTMLUListElement | null {
  const list = item.querySelector(':scope > ul');
  return list instanceof HTMLUListElement ? list : null;
}

/**
 * Retourne les enfants directs d'une liste.
 *
 * @summary Récupère les `<li>` directs d'une liste `<ul>`.
 * @param list Liste concernée.
 * @returns Liste des nœuds enfants directs.
 */
function getDirectItems(list: HTMLUListElement): HTMLLIElement[] {
  return Array.from(list.children).filter(
    (element): element is HTMLLIElement => element instanceof HTMLLIElement,
  );
}

/**
 * Simule un drag and drop simplifié du nœud source vers une cible.
 *
 * Si `target` vaut `null`, le drop est simulé sur l’arbre lui-même,
 * ce qui correspond à un drop à la racine.
 *
 * @summary Simule un déplacement par drag and drop.
 * @param source Nœud déplacé.
 * @param target Cible de drop ou `null` pour la racine.
 * @param clientY Position verticale du curseur pour le drop.
 */
async function simulateDrop(
  source: HTMLLIElement,
  target: HTMLLIElement | null,
  clientY = 50,
): Promise<void> {
  const dataTransfer = {
    effectAllowed: '',
    dropEffect: '',
    setData: vi.fn(),
    getData: vi.fn(),
  } as unknown as DataTransfer;

  source.dispatchEvent(
    new DragEvent('dragstart', {
      bubbles: true,
      cancelable: true,
      dataTransfer,
    }),
  );
  await flush();

  target?.dispatchEvent(
    new DragEvent('dragenter', {
      bubbles: true,
      cancelable: true,
      dataTransfer,
      clientY,
    }),
  );
  await flush();

  target?.dispatchEvent(
    new DragEvent('dragover', {
      bubbles: true,
      cancelable: true,
      dataTransfer,
      clientY,
    }),
  );
  await flush();

  target?.dispatchEvent(
    new DragEvent('drop', {
      bubbles: true,
      cancelable: true,
      dataTransfer,
      clientY,
    }),
  );
  await flush();

  source.dispatchEvent(
    new DragEvent('dragend', {
      bubbles: true,
      cancelable: true,
      dataTransfer,
    }),
  );
  await flush();
}
